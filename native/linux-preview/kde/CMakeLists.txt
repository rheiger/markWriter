cmake_minimum_required(VERSION 3.22)
project(markwriter_thumbnail LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ECM 6 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6KIO REQUIRED)
add_library(markwriterthumbnail MODULE thumbnail.cpp render.cpp)
set_target_properties(markwriterthumbnail PROPERTIES PREFIX "")
target_link_libraries(markwriterthumbnail PRIVATE Qt6::Widgets KF6::CoreAddons KF6::KIOGui)
install(TARGETS markwriterthumbnail DESTINATION ${KDE_INSTALL_PLUGINDIR}/kf6/thumbcreator)
include(CTest)
if(BUILD_TESTING)
  add_executable(render_test render_test.cpp render.cpp)
  target_link_libraries(render_test PRIVATE Qt6::Widgets)
  add_test(NAME render_test COMMAND render_test)
  set_tests_properties(render_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
if(BUILD_TESTING)
  add_executable(kio_test kio_test.cpp render.cpp)
  target_link_libraries(kio_test PRIVATE Qt6::Widgets KF6::KIOCore KF6::KIOGui)
endif()
